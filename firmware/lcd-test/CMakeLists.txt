cmake_minimum_required(VERSION 3.13)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(gba_lcd_test C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(gba_lcd_test main.c)
target_link_libraries(gba_lcd_test pico_stdlib hardware_spi hardware_pwm hardware_i2c)
pico_enable_stdio_usb(gba_lcd_test 1)
pico_enable_stdio_uart(gba_lcd_test 0)
pico_add_extra_outputs(gba_lcd_test)
